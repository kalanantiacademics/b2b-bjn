Yang Bisa Kita Lakukan

insert() menambahkan item.

delete() menghapus item dari tampilan.

curselection() mengecek item mana yang sedang dipilih user.

get() mengambil teks dari item terpilih.

def load_contacts():
    # Bersihkan listbox sebelum memuat data baru
    listbox.delete(0, tk.END)

    # Baca file dan masukkan ke dalam listbox
    with open(FILE_NAME, "r") as file:
        for line in file:
            if "," in line:
                nama, nomor = line.strip().split(",", 1)
                # Tampilkan di layar dengan format "Nama - Nomor"
                listbox.insert(tk.END, f"{nama} - {nomor}")
